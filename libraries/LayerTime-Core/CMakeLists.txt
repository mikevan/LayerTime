cmake_minimum_required(VERSION 3.16)
project(LayerTimeCore LANGUAGES CXX)

# Consumers may supply their own compatible sensor target, or an explicit
# checkout. No download or implicit sibling-directory lookup occurs here.
if(NOT TARGET LayerTimeSensors)
    if(NOT EXISTS "${LAYERTIME_SENSORS_DIR}/src/lts/Detectors.h")
        message(FATAL_ERROR "Set LAYERTIME_SENSORS_DIR to a LayerTime-Sensors checkout, or supply target LayerTimeSensors.")
    endif()
    file(GLOB SENSOR_SOURCES CONFIGURE_DEPENDS "${LAYERTIME_SENSORS_DIR}/src/lts/*.cpp")
    add_library(LayerTimeSensors STATIC ${SENSOR_SOURCES})
    target_include_directories(LayerTimeSensors PUBLIC "${LAYERTIME_SENSORS_DIR}/src")
    target_compile_features(LayerTimeSensors PUBLIC cxx_std_17)
endif()

file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/core/*.cpp")
add_library(LayerTimeCore STATIC ${CORE_SOURCES})
add_library(LayerTime::Core ALIAS LayerTimeCore)
target_include_directories(LayerTimeCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(LayerTimeCore PUBLIC cxx_std_17)
target_link_libraries(LayerTimeCore PUBLIC LayerTimeSensors)

option(LAYERTIME_BUILD_TESTS "Build the standalone core unit suites" OFF)
if(LAYERTIME_BUILD_TESTS)
    enable_testing()
    if(NOT EXISTS "${LAYERTIME_CONTRACTS_DIR}/vectors/link_frames.json")
        message(FATAL_ERROR "Set LAYERTIME_CONTRACTS_DIR to the shared LayerTime-Contracts checkout for codec conformance tests.")
    endif()
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test/check.h")
        message(FATAL_ERROR "Standalone tests are included in the prepared LayerTime-Core repository. Workspace integration tests remain under LayerTime/test.")
    endif()
    file(GLOB CORE_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/test/test_*/*.cpp")
    foreach(test_source IN LISTS CORE_TESTS)
        get_filename_component(test_name "${test_source}" NAME_WE)
        add_executable(${test_name} "${test_source}")
        target_include_directories(${test_name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/test")
        target_compile_definitions(${test_name} PRIVATE LAYERTIME_LINK_VECTORS_FILE="${LAYERTIME_CONTRACTS_DIR}/vectors/link_frames.json")
        target_link_libraries(${test_name} PRIVATE LayerTimeCore)
        add_test(NAME ${test_name} COMMAND ${test_name})
        set_tests_properties(${test_name} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/test")
    endforeach()
endif()
